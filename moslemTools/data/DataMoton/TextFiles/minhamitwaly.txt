بسم الله الرحمن الرحيم
(مِنْحَةُ  مُولِي   الْبِرِّ)
فيما زاده كتاب النشر للقراء العشرة على الشاطبية والدرة
للعلامة  محمد محمد هلالي الأبياري       -رحمه الله-
*********************************************
المقدمة
1	قالَ محمدٌ هلالي راجيًا
	إلـٰهَهُۥ عفوًا عميمًا كافيًا
2	حمدًا لموْلانا مصليًا على
	محمدٍ والآلِ ما تالٍ تَلا
3	وهاك ما للكلِّ نشرٌ زادَهْ
	عما بدرةٍ وحرزٍ سَرَدَهْ
4	وما مِن الخلافِ ها هنا يحلّ
	ففيه وجهٌ من كليهما قُبِلْ
5	وآخرٌ مما يزيدُ النشرُ
	ومنه جا بالأصبهاني الذِّكرُ
6	وَهْوَ لورشنا طريقٌ يُقبَلُ
	وأزرقٌ له طريقٌ أوَّلُ
7	فإن تركتُ ذِكرَ الاصبهاني
	فهْوَ وأزرقٌ موافقانِ
8	وإن لبعضِ ما لأزرقٍ سكتْ
	عنهُۥ يكن موافقًا فيما ثبتْ
9	ممارسًا فيما أقول الطيّبة
	مُتَّبِعًا رموزَها المهذّبة
10	مقتصرًا على الذي به قُري
	ومهملًا ما رَدُّه لنا دُري
11	وكل ما بالضعفِ من حرزٍ  وُصِفْ
	ذكرتُه إن كان من نشرٍ أُلِفْ
12	سميته: ( مِنْحَةَ مُولِي البِرِّ )
	بما يزيده كتابُ النشْرِ
13	فقلت راجيًا إله الخلقِ
	هدايتي إلى طريق الحقِ
*********************************************
البسملة وسورة أم القرءان والإدغام الكبير
14	بسمَلَ بين السورتين كم حِما
	والاصبهانيُّ كقالونَ افهما
15	واسكتْ لبزارٍ    صرٰطَ كُلَّهُ
	بالصادِ زُرْ ، ومحِّضًا أوَّلَهُ
16	أو محِّضًا وأشْمِمًا في الثانِ أوْ
	ذي اللامِ عن خلادِهم كما رَوَوْا
17	وبابَ أصۡدقُ بخلفٍ غِثْ  وما
	يُدغَمُ خلفَ السوسِ والدورِ اعلمـا
18	وعندَ مدَّ الفصلِ أو تحقيقِ
	همزٍ فلا إدغامَ بالتحقيقِ
19	والميمَ والبا رُمْهُمَا ولا تُشِمْ
	وامنعْهما في الفا بفا لبعضهِمْ
20	ورجَّحوا إدغام غيثٍ في جعل
	بالنحلِ مع  ذهب وأيضًا لا قِبَل
21	وأنه بالنجم أخراها . وزِدْ
	خلفًا على الذي بـ(درةٍ) وُجِدْ
22	في با العذابَ ، مِن جهنَّمَ معَ ـا
	مبدلَ الكهفِ وفي لِتُصنعَ ـا
23	والكافِ في كانوا وكلا ، أنزلَ ـا
	لَكم ، تمثلَ لها ، و جعلَ ـا
24	شورى ، وعنه البعض في جَعَلَ عَمّ
	وقيل مثلُ ابنِ العَلا يعقوبُهمْ
25	والياءُ في والَّـٰىۡ مع يئسنَ ـا
	إدغامها هدايةٌ حفَّتْنَا
*********************************************
باب هاء الكناية
26	واقصرْ  يؤدهۡ  ، نؤتهۦ ،  فألقهۦ
	نصلِهۡ  ، نولِّهۡ  مِن ثَنًا   يتقِهۦ
27	ذُق مِز ، وصِل خُذ  يرضَهُۥ ذِع ، واقصرَنْ
	مِز خُض ، وسكِّنْها صَبا  والكلَّ لنْ
28	مع لم يرهۡ  وحرفيِ الزلزالِ خذ ،
	قصرَ الثلاثِ خَف ظَمًا  أرجـٔۡهُ لـُذ
29	وشعبةٌ فيها كبصرٍ ،  و صِلَا
	خذ  يأتهۦ غيثٌ يلي ، واقصر خلا
30	وترزقانهۦ  بدا ، صِل خيرها
	والاصبهانيُّ بهِ ٱنظُر ضمَّ  ها
*********************************************
باب المد والقصر
31	إنْ يَنْفَصِلْ فالقصرُ لِـي عُدْ ، مُدَّ ظِلّ
	يُمْنٌ ، وأشبِع مِزْ  والِاتصالَ كُلّ
32	ومَدَّ للتعظيم كُلُّ مَن قصَرْ
	عينَ اقصرنْ للكلِّ  ـذَيۡنِ  ـتَيۡنِ دَرْ
33	واللينَ غيرِ لفظِ شىۡء جدِّدَا
	وعنه إسرٰٓءيل وسِّط وامْدُدَا
34	كـ لا مَرَدَّ الوسطَ  مع شىء   فُلا
	والاصبهانيُّ كقالونَ تَلا
*********************************************
باب الهمزتين من كلمة
35	وحققنْ أىِٕنكم الَانعامِ غَرْ
	وسهِّلَنْ ءأسجد الإسرا مَقَرْ
36	ومُدَّ واقصر مُسْجَلًا لبَّى  ، ولا
	يقصرُ ما بفصلتْ إنْ سهّلا
37	وقبلَ ضمةٍ بقصرٍ بانِي
	والفتحَ لا تبدلْ للَاصبهانِي
38	ءامنتمُ  أخبر له  ، تحقيقُها
	لـِي ، واسأَلَنْ طه ، وحقِّقْ مُلْكَــها
39	الَاعرافَ وصلًا    زُرْ  وسَلْ ءا۬عجمِى
	لـنا   ، وأخبِرَنَّها غيثٌ ز كِي
40	وامْدُدْهُ مع أن كان مِز   وأبدَلوا
	أئمة كلًّا لمن يُسَهِّلُ
41	ومُدَّ سَهِّلنْ  للَاصبهانِي
	في سجدةٍ وما بقصٍّ ثانِي
*********************************************
باب الهمزتين من كلمتين
42	الُاولَى اسْقطنْ إنْ وافقا زاهٍ  غَلا
	والاصبهاني ثانِ ذا لن يبدِلا
*********************************************
باب الهمز المفرد
43	يُؤَيُّدُ الإبدالَ خذ  وأبدلا
	بالخلف فيما يبدِلُ السوسي حَلا
44	والمؤتفكـ كلًا  بدا  نَبِّئۡنَا
	ثِق  الَاصبهاني مطلقًا  لا جِئۡنَا
45	نبّأتُـ   ، هيِّئۡ ،  لؤۡلؤٗا وكأسُ
	تُـٔۡوِيهِ ، تُـٔۡوي ، الرأسَ ، رِئۡيٗا ، بأسُ
46	الِاقرا  مؤذن ،  لِئَلَّا وابدلِ
	ناشئةَ ، الفؤادَ ، خاسئٗا مُلِيـ
47	بِأَيِّ ذي الفا واختلفْ سواها
	وسَهِّلَنْ بقصصٍ رءاها
48	كذا رأيتُهم ، رأيتُ يوسفَا
	رأتهُ مع رءاهُ نملٍ وُصِفَا
49	رأيتَهم تعجِبـُ  مع أخرى اطمأنّ
	وأفأنت و كأن ، أفأمن
50	لأملأن ، أفأصفـٰـ   ، ويكأن
	تأذنَ الأعرافِ، والخلفُ اسْتَكَن
51	في إِبْرَهَمْ وفي النسِيءُ  اهمز ولا
	تبدِل له أريتمُۥ بل سهِّلَا
52	وادغم هَنيٓئٗا  وبَرِيٓـٔٗا  و مَرِيـٓ
	ثَبْتٌ  كَهَيۡئَة لّه فأظهِرِ
*********************************************
باب النقل والسكت على الساكن وغيره
53	ٱلَـٰنَ في الإخبار بالخلف خطِف
	والاصبهاني معه في مِلءُ اختُلِف
54	وانقل بواوٍ عادٗا الۡأُولَى بَهر
	وبالذي لـخلفٍ  في السكت قرّ
55	أو معَ موصولٍ فدًا ، وبعضهم
	في غير شيءٍ  ، أو بلا سكتٍ يعمّ
56	أو عكسُ ذا ولو يكونُ حرفُ مد
	وغيرُه إدريسُ مع مَولًى عَمَد
57	وتركُه في عوجٗا ، مرقدنا
	بل ران ، من راقٖ بنَصِّ حفصِنا
*********************************************
باب وقف حمزة وهشام على الهمز وإدغام ذال إذ ودال قد وتاء التأنيث
58	وسهِّلَنْ لـحمزةٍ همزًا حصَل
	في البدء إن بكلمةٍ قبلُ اتصَل
59	وسهِّلَنْ عن ألفٍ ومُدّا
	واقصر ، وعن واوٍ وياءٍ مدّا
60	فانقِل ، وأدغم وهْو أقوى في الصلة
	والنقلُ عند ميمِ جمعٍ أهمَلَه
61	ولِـهشامٍ حَقَّقَنْ في الطَرَفِ
	وأظهِرَنْ  إذ عندَ دالٍ مُنصِفِي
62	وأدغِمَنْ قال لقد في صَادِها
	مع ـهُدِّمَت والتاء في سَـ جَـ زْ  لـَها
63	و أَنبَتَت مِز  عنه في الثا أظهرا
	والتاء في الظا الاصبهاني أظهَرَا
*********************************************
باب إدغام لام هل وبل
64	وخلفُ بل طبعَ فُز  وكلُّها
	لا  الرعدِ ، مع نونٍ وضادٍ لُطفُها
*********************************************
باب إدغام حروف قربت مخارجها
65	بالجزمِ في الفا الخلفَ لـُذ قُم عذتُ
	ـنَبَذۡتُـ لَنْ  والاتخاذ غِرْتُ
66	أورثتُـ  مِزْ   يس ، ن والقلم
	نَل مِن هُدًى إِذًا   يعذب من بَسَمْ
67	دُم فائزًا  يلهث نَدًى جُودٌ  لَنا
	ثِق دائمًا  و اركب نَداه زُهدنا
*********************************************
باب أحكام النون الساكنة والتنوين
68	ـيُنغِضـ    ، يَكُن ،  ـمُنخَنِقـ   ٱخْفِ ثِق  وغُن
	لامًا و رَا  لا صحبةَ  اليا دَع تَصُن
*********************************************
باب الفتح والإمالة وبين الفظين
69	مَيِّل ـأُوَ ٰرِى وكلا يُوَ ٰرِى
	تُمَارِ تُب  وخلف ـغَارِ ، البَارِى
70	عين اليتـٰمى و النصـٰرى مُسْجَلا
	كذا أُسَـٰرَى و سُكـٰرى وصَّلا
71	كذا كُسالى   عنه  والخلافُ في
	هارٖ بَدا  خابَ ، مَشاربُ كُفِي
72	حرفَيْ  رءا  و  زاد ، شا ، جا ، ءَانِيَة
	إِنَىٰهُ ، عابدونَ ، عابدٞ  لِـيَه
73	يلقىٰه   ، مُزۡجَىٰةٖ و ـشَـٰرِبِينَ
	ذِي الرا ،  أَتَىٰ أَمۡرُ ، الحواريِّـۧنَ
74	وقبلَ را  كَسرٍ  و كـٰفرين مع
	مكررٍ مَنٌ ، و فتْحُه قَنِع
75	والمَيْلَ فِد  والخلفُ في يَـٰبُشۡرَىٰ
	رمى ، بلى ، نونِ نـٔا بالإسرا
76	سِوٗى ، سدًى ، أدرىـٰـ ، رءا  لا أَوَّليْـ
	ـهِمَا صَبا   والجارِ ، جرُّ  الناسِ  طَـيْ
77	مع   ـأسفى و ـحسرتى و ـويلتى
	أنى   وخُلْفُهُ  عسى ، بلى ، متى
78	وخُلْف (فعْلَى) و رءوسِ الآي .  لا
	ذي الراءِ  حز  وميَّلِ الدنيا طِلا
79	وخلفُ إدريسَ بـرؤيا غيرَ (ال)
	ـقهارِ والبوارِ بالفتح فَصَلْ
80	يا كاف لـي  -ها يا- إذًا  يا، حا حَلا
	يس قلِّل فِد إذًا  طه جَلا
81	والمَيْلَ في التورىٰة فِد مهما يَحِل
	وغيرَها للاصبهاني لا تُمِل
82	وما يُمَالُ افتح  وقلِّل إن سكن
	إن كان للإدغام أو  وقفٍ   يمَن
*********************************************
باب إمالة هاء التأنيث وما قبله في الوقف
83	وبعدَ  أه ، و عشرِها، فِطۡرَتَ رُم
	خُلفٌ  وقيل مثلُه حمزَتُهم
*********************************************
باب الراءات واللامات
84	للازرقِ الخلافُ في  مِرَاءٗ
	و ـشَرَرٖ ، إِجرامِـ  و  افتراءً
85	عشيرتـُ  التوبةِ مع  ذراعٗا
	وِزۡرَ ، ذِرَاعَيۡهِ ، إرمۡ ، سِراعا
86	تنتصران ، حصرت  و  وِزركَ
	و كِبرَه ، لَعِبرة ، و ذِكركَ
87	الِاشراق ، ـسـٰحِران مع  أن طهِّرا
	و حذركم وإن يَصِل كـشاكرا
88	خيرًا  وذاتَ الضم رقِّق في الأصح
	والخلفُ في عشرون مع كِبۡرٞ وَضَح
89	ولام صَلصـٰلٖ وعن طاءٍ و ظا
	والاصبهانيُّ كقالونَ عَظَا
*********************************************
باب المرسوم
90	هيهاتَ قِف بالهاء زِن  واختلَفا
	في نحوِ مُوفُونَ سِنِينَ ظَرُفَا
91	واقتده اقصُر مِن  ويا وادِ احذفِ
	بالنملِ رُصْ   بِهَـٰدِ رُومٍ راقَ فِ
*********************************************
باب ياءات الإضافة
92	بالخلفِ ما لِى الطَولِ مِز ، يس لـِي
	والنملَ لـِي خُذ  يـٰعبادۦ لا  غَـِلي
93	لِى نعجةٞ رهطِى لِوًى  و أَنِّىَ
	أُوفِى ثَنًا  عِندِى  بقَصٍ داعيًا
94	وسَكِّنَنْ للأصبهانيِّ  لِىَ
	فيها  و   إخوتِى وفي  أوزعنِىَ
95	في النملِ والأحقافِ مَحۡيَاىَ بِلا
	خُلفٍ ، و فتحُهُۥ  ذَرُونِى حُصِّلا
*********************************************
باب ياءات الزوائد
96	دُعَاءِ  ، مَنۡ يَتَّقِ  ، نَرۡتَعِ   اختُلِف
	مع وَقْفِ  ءاتىٰنِ  زَها كِيدُونِ لِف
97	ـبَشِّرۡ عِبَادِ يـا ،  ـعِبَادِ فاتقو
	غِثِ   التلاقِ و التنادِ بارِقُ
98	و الاصبهانيُّ كأزرقٍ  و عنه
	اتبعونِ أهدكم  و  إن تَرَن
*********************************************
فرش الحروف: من سورة البقرة إلى أول سورة المائدة
99	يملَّ هۡو ، ثُمَّ هۡو خلفٌ ثِق بِنَص
	قبلَ اسجدوا  شمَّ  ـالملـٰىِٕكة خَص
100	خُطۡوَٰتِ هَبجُرۡفٍ لوَىخُشۡب زَهَد
	ـسُحۡقٗا رَسا خيرٌ و  يُسر الذَّرْوِ خَد
101	وبابُ يأمركم بالِاختلاس يَد
	الِاتمامَ طِب  و جَبر ِٔيل اليا صَعِد
102	ميكـٰىِٕلَ احذِف زِن  و إِبۡرَ ٰهَـٰمَ مَن
	مهما أتى  ننسخ بفتحتيه لَن
103	أرنا وأرنِى اسكِن طِبِ ، اختلِس يَلِي
	فصلتِ اكسِر لُـذ يَرىٰ خاطِب خَلِي
104	في الساكنين الخلفُ في التنوين مَر
	والجرَّ زُر  معًا    يُضَار الثِّقْلُ ثَر
105	بالخلفِ يَبصُۜط ، بَصۜۡطَةٗ زُر مَن يَفِي
	عُد  بسطة      العلمِ زَها  وخَفِّفِ
106	تاءً لبزٍ شُدِّدَتْ وَصْلًا  وفي
	كِلا نعما سَكِّنَنْ حُز بِن صَفِي
107	هـٰأنتمۥ للأصبهانِي مُسْجَلا
	لاتُبْدِلَنْ  ،  وأَثْبِتَنْ زُهدٌ جَلا
108	ـما يفعلو  ، لن يكفروه غِب طلَع
	ما قُتِّلُوا خَفِّفْ  ، وبا ـٱلۡكِتَـٰبِ دَع
109	لنا  وخاطِب يُظلمو    شِد  مؤمِنا
	الُاخرَى افتحَنْ ثِقسكِّنَنْ تَعدُوا بِنا
*********************************************
من سورة المائدة إلى أول الروم
110	شنئانُ حرِّكْ ذُق  و رِضوانـ  اضْمُمَنْ
	ثانٍ وذكِّر لم يَكُن صُن إن يَكُن
111	لـُذ  خِفَّ مع تحتُ فتحنا ذُق غَرَرْ
	واقتربت غَرْ واكسِرْ اضطرر  خَبَرْ
112	و المعز سكِّنْهُۥ ، ويا بيس لَسَنْ
	أن لعنةُ اشدُد ناصبًا ، حـىّ اظهِرَنْ
113	بالكسر زِن  وضم يعكفون عَنْ
	إدريسَ  يا وَلِـِّۧىَ الُاخرَى احذفَنْ
114	وافتحه واكسر يُسرَه  لا يحسبن
	كالنور عن إدريسَها يهدى افتحن
115	حُز، واخفِ ذُق، واسكِنْ بَقِي وفَٱجۡمَعُواْ
	خلفٌ غَدا  ذكِّرْ تكونَ صَنَعوا
116	والنونُ في     تَتَّبِعَانِ خُفِّفَا
	تَسۡـَٔلنِ مَا  بالفتح لِـي واختلَفَا
117	كلُّ ـيضِلُّوا ، يُلهِهم ، و يُغنِهم
	قِهموفي  ادخلو    انقلنْ مع كسرِ ضَم
118	غَر ـيجزين نونٌ ويا كَم وافتحوا
	خِطۡـٔٗا بتحريكٍ لَـنا  يسَبِّحُ
119	عمَّا يقولو   الخلفَ غِثْ  ءَاتُونِى
	اقطع ،  لَّدُنِّى رُمْ ، تسـٰقط  صَوْنِى
120	وبالخلافِ ٱشۡدُدۡ ، وأَشۡرِكـۡ ، يَأۡتِهِم
	خُذيَصِفُو   مِزواجمعَنْ في الريح ثَم
121	أَذِنَ عَن إدريسَ ضُمَّ  وارفعَن
	عَـٰلمِ بدءًا غَر ورأفة سكَن
122	هَب، في الحديدِ حرِّكًا وامدُدْ زَهَوْا
	واكسِرْ جيوبـِ    صُنْ يقولو  زَنْ يَرَوا
123	كيفَ صَبا وحـٰذِرونَ الخلفَ لَم
	ما يفعلو  كَم صِف و يعقلونَ يَم
*********************************************
من سورة الروم إلى سورة سبأ
124	ـنذيقهم بيائهِ خُلْفٌ زَرع
	واقصر ـأَتَوۡهَا مِز كثيرٗا با لـَمَع
*********************************************
سورة سبأ وأختيها
125	مِنسَأته الِاسكانُ لِـي يَنقُصُ ضُم
	وافتح غِنًى يا يَخصِمون الكسرَ صُم
126	والخاءَ لـُذ ، وسكنَنْ بِن ، وافتحَن
	حُرٌ  بَدا ـلا يعقلون الخلفَ كُن
*********************************************
من سورة الصافات إلى أول الفتح
127	للاصبهاني سكنَنْ بالنقل أَوۡ
	ءاباؤنا  ، عنه اصطفى وصلًا رَوَوْا
128	إلياسَ صِل ،  ـخَالصةٖ نونٌ لَـيَ
	ولا تزِد نونًا  أ تأمرونِىَ  ،
129	يدعونَ خاطِب مِز وقلبِ نوِّنَن
	بالخلفِ كَم سيدخلون سَمِّ صُن
130	ما يفعلو  غِث خلفَ  يُرۡسِلَ ارفعَا
	ـيُوحِى اسكنَنْ مِز يا  نُقَيِّضۡ صَانِعَا
131	و ءانِفًا ، لِينذرَ الخلافُ هَب
	كَرهٗا بضمٍ ،   لِنُوَّفِّـ   النونُ لَب
*********************************************
من سورة الفتح  إلى   أول الحديد
132	والخلفُ في  ءَازَرَهُۥ  لَـدينا
	و ما أَلِتۡنَـٰـ    همزَه احذِف زَيْنَا
133	ـمُصَۜيطرون السينَ مِز ، والصادَ زِد
	وضُمَّ  لم يَطمثـ   معًا بالخلفِ رِد
*********************************************
من سورة الحديد إلى أول المعارج
134	نَزَّلَ خَفِّفْ غِث   يكونَ ذَكِّرَنْ
	دُولَة انصِبْ ، خِفَّ يفصلُ لَسَنْ
*********************************************
من سورة المعارج إلى أول الغاشية
135	لا يَسۡـَٔلُ اضمُم هَب  وذكِّر تُمۡنَىٰ
	لا نونَ في سلاسلٗا لَـدينا
136	نَوِّنْهُ غِث ، وامدُدْهُ وقفًا زِن غَرَرْ
	واقصره معْ أُولَى قواريرَا شَكَرْ
137	والثانِ لُـذ  خاطِب يشاءون كَرَهْ
	وأُقِّتَتۡ شُدَّ اهمِزَنْ ذُقْ   نَـٰخِرَة
138	قصرٌ تَلا   وثِقْلُ سُجِّرَت غَلا
	و سُعِّرَت صِف  فَـٰكِهين اقصُر كَـلا
*********************************************
من سورة الغاشية إلى أول العلق	
139	ـمُصۜيطرٍ بالسين زِن مَـيُّ عَطَف 	
	وبعدَ بَل لا أربعٌ خاطِب شَغَف
*********************************************
من سورة العلق إلى آخر القرءان الكريم	
140	و أن رءاه  اقصُرهُ وامْدُدْ زمرةُ	
	و النَّـٰفِثَـٰتِ بالخلاف غايةُ
141	وها هنا تمامُ نظمِ الِمنْحَة	
	بِحَمْدِ مَوْلَانَا مُفِيضِ النِّعْمَة
142	أبياتُهُ يُمْنٌ جَلَيٌّ أُرِّخَتْ	
	لِنُصْحِنَا نُصُوصُهَا تَدَوَّنَتْ
143	فيا إلـهي انفع به مَنْ رَامَهُ	
	واجعلْهُ مَقْبُولًا وسهِّلْ فَهْمَهُ
144	وصَلِّ دائمًا على محمدِ	
	وآله وصحبهِ الأماجدِ
