الفوائد المعتبرة في القراءات الأربع بعد العشرة
للإمام محمد بن أحمد المتولي
*******************
المقدمة
1 قال محمـــــــد و هو ابن أحمدا
المتـــــــولي رب كن لي مسعدا
2 أحــــــمد ذا الجلال و الإكرام
سبـــــــحانه جل عن الأوهام
3 و أفضل الصــــــلاة و التسليم
لذي المــــــقامات العلي الكريم
4 نبـــــــــينا الأمي ثم عترته
و صــــــحبه من اصطفوا لرؤيته
5 و بعد فخذ نظمــــي حروف أربعة
زادت علي العشـــــر و كن متبعه
6 فابن محيــــــــصن هو المكي
أولهم فالأعمــــــــش الكوفي
7 و الشنبــــــوذي روي علي سند
عنه كذا مطــــــــوعي استند
8 ثم من البــــــــصرة الآخران
الحــــــسن السامي و يحيي الثاني
9 جعلـــــت أصل ابن كثير يا فتي
للمك و الكــــــوفي أصل حمزة
10 ثم للآخـــــــــرين قد تقررا
أصل أبي عمــــــرهم كما تري
11 فحيــــــثما قد خالفوا ذكرت لا
ما اتــــــــفقوا عليه في الحرز
12 و جيم مبـــــــهج و فا مفردة
إشارة المكي و مـــــــيم عمت
13 ثم الألف مع شينــــها و الطاء عن
كـــــوف و راوييه و الحا للحسن
14 أما الــــــيزيدي فبلا رمز وجد
لقلة انـــــــــفراده فيما يرد
15 سميته الفـــــــــوائد المعتبرة
فأســــــــال الكريم أن ييسره
16 و ربنا المأمـــــــول في القبول
بجاه طه المـــــــصطفي الرسول
*******************
باب الاستعاذة و البسملة
17 زد الســــــميع و العليم قبل من
(حز) بعــد إن الله هو (حصن) ( أمن )
18 و ادغما ( حما ) (شفا) و بســــملا
(طب) فاصلا  و عند بصـــري صلا
19 و لليزيدي الســـكت زد و للحسن
في بدء غير الحمد لا تبســـــملن
*******************
سورة أم القرآن
20 الحمد لله بكـــــــسر حيث جا
(حز)مالك انصب(طب) ومد(طب)(حجا)
21 نعبد ضـــم افتح بيا ( حز) و اكسرا
نون و تاء من مـــــضارع (طرا)
22 إن عين ماضـــــيه الثلاثي كسرت
و هي بفتح في مــــــضارع أتت
23 أو زاد ماضــــــيه علي الثلاثة
و فيه همز وصــــــل في البداءة
24 كنستـــــــعين تيئسوا تذر وقر
وجهان في تضـــحي و تطغوا مع تقر
25 سراط كلا ( فز) فقــط سراط (شم)
وصاده مع أل و مطــــــلقا أشم
26 ( طب) و صراطا مستقيـــما ( حليا)
و ميم جمع بعد كســـــر صل بيا
27 و بعد ضــــــمة بواوها ( حتم )
و غير بالنــــــصب (جمالة) وسم
*******************
باب الإدغام الكبير
28 أدغم في الباب الـــــيزيدي كأبي
عمرو علي الخلاف  فافــــهم تصب
29 والاه في إدغامه المـــــثلين ( حم )
(طب) (فز) و (جيده) إذا الأول ضــم
30 و البا ببا (شفا) مناسكــــكم و ما
سلكــــكم (فز) (طيبا) و زد ( حما)
31 يحزنك مع تاء الضمير مسجـــــلا
و (طب) بمثلي كلمـــــة لا التا تلا
32 و أتحاجــــوننا ( فتي ) (طب) أدغما
و في بأعـــــــيننا بطور عنهما
33 هذا و والي المـــــك في قرب علي
قاف بكاف إن بكـــــــلمة بلا
34 خلف كذا أخرج شـــــطأه و في
ميم ببا مع يعـــــذب من ( شفي )
35 كذاك في تصـــــــلية المطوعي
كذا بباقـــــي الباب ( فاضل) يعي
36 و زد وعظــــت مع إطباق ( متي )
و الـضاد في الطا ( مز) و في التا (فاثبتا)
37 و ابن محيـــــــصن بإظهار تلا
جميع ما فيه اخـــــتلاف ابن العلا
*******************
باب المد و القصر
38 وسط لهم مدا و قصـــــر المنفصل
لحســـــــن و ابن محيصن نقل
39 ثم اليـــــــــزيدي بخلفه تلا
و الشـــــــنبوذي بإشباع كلا
*******************
باب الهمزتين من كلمة
40 سوي آلهتنا حــــــــقق ( حما)
و في جــــــميع الباب قصره سما
41 و قبل ضم للـــــيزيدي اقصر ولا
إبدال في تبارك الملــــــك ( ملا)
*******************
باب الهمزتين من كلمتين
42 أسقـــط ( فتي ) حال اتفاق و ( جلا)
فتحا و أولي الكـــــسر عنه سهلا
43 لكنه بالـــــــسوء إلا ما رحم
له بإدغام و تسهـــــــيل وسم
44 له بأخري الضــــــم ثم للحسن
حال اتفاق و اخــــــتلاف حقق
*******************
باب الهمز المفرد
45 ســـؤلك أبدل (شم) و كالأرض ائتيا
(مضي) و أنبئهم و نبئــــهم (حيا)
46 و اكسر و ها أنتم بتســـــهل له
و قل لئلا أعمــــــــش أبدله
47 و اللاء ســهل (مز) و باليا اهمز (حما)
و عنه باقـــــــي الباب همزه نما
48 و قد روي يحــــــيي جميع الباب
مثل أبي عمـــــــرو بلا ارتياب
*******************
باب نقل حركة الهمزة إلي الساكن قبلها و السكت
49 و نقل آلان و مــــــعي ردا فهم
و أقرأ بترك الســـــكت باتفاقهم
*******************
باب وقف الأعمش علي الهمز
50 قف عنه بالتحـــــقيق أو كحمزة
و الخلـــــف في الأول أيضا اثبت
*******************
باب الإظهار و الإدغام ذكر ذال إذ
51 إذ أدغـــم المكي و غير الجيم (حل)
صـــفيرها فقط (أتي) و الجيم ( طل)
ذكر دال قد و تاء التأنيث و لام هل و بل
52 للكل قد و التاء أدغمــــن  و هل
و بل ( مضي) لكن بنــون هل ( جعل)
53 بل تؤثـرون (حز) و (طب) في الطا فقط
و الباب بالإظـــهار (شم) بلا شطط
*******************
باب حروفت قربت مخارجها مع أحكام النون الساكنةو التنوين
54 بالجزم يلـــــهث من يرد  أورثتموا
لبــــــثت و اتخذت صاد أدغموا
55 لهم و في نبذتها مع عــــذت ( فن)
و الرا بلام معه يحـــــيي لا الحسن
56 و اركــب سوي ( فتي ) و يس ( أثر)
(مـــدا) و في نون ( شفاها) (فاعتبر)
57 طـــــس ميم ( شم) و غنة سقط
و في وي لدي مـــــطوعيهم فقط
58 و أظــــــــهرن ثلاثة رابعهم
(فتي ) وأدغم خمســـــة سادسهم
59 أن سيــــــكون منكم مرضي مية
سنين مع يومئذ ثمانيـــــــــة
60 كـــــــذلك أزواجا ثلاثة و في
ثجاجا أيضا لا غــــــــنة قفي
*******************
باب الفتح و الإمالة
61 بوار قهار للأعمـــــــش افتحن
و عنه آتيــــــك ضعافا أضجعن
62 أجـــــاءها له أضاء ( طب ) كذا
ضارين مع نون نأي افتحــها( شذي)
63 توراة عن يحيي و أعـــــمش أمل
و للــــــيزيدي هذه أعمي نقل
64 راها فواتح كذا هـــــــمز رأي
مع ألــــــفات بعد راء قد رأي
65 و باب را كســــر سوي الجار قرا
وصلا و مع الأعمــــش فيما كررا
66 و كــــيف كافرين يحيي و اختلف
في الناس و افتـــــح غير ما وصف
*******************
باب الوقف علي أواخر الكلم
67 و وقفهـــــــم بالروم و الإشمام
أجز و الأعــــــمش بنص سامي
*******************
باب الوقف علي مرسوم الخط
68 هــيهات قف بالهاء ( جد) و (فز) بتا
فان و راق مع ينـــــاد اليا ( متي )
69 صل يتســـــنه دون ها لا للحسن
كذا اقتـــده لا ( جد) كتابيه ( منن)
70 حســـــــابيه و ماليه و سلطانيه
له فقط و غير يـــــــحيي ماهيه
71 و زاد حــــذفها لدي الوقف ( فلا)
وقـف بكاف ويك ( فز) و اليا ( طلا)
72 أيا و مال أو بما للـــــــكل قف
و نحو فيـــــم عم عنهم ها حذف
*******************
باب مذاهبهم في ياءات الإضافة
73 و قبل همز القطع أســــكن للحسن
إلا و يـــــسر لي معي أو فافتحن
74 و ابن محيـــــــصن كبزي خلا
إني أراكم مـــــــع و لكني كلا
75 و تأمـــــروني أدعوني عندي فطرن
فاسكـن و أجري فافتح له و فتح (فن)
76 إني الأخــــــيرتين في العقود عن
و عــــند لام العرف للمكي اسكنن
77 وافـــــق ( حز) لا الأنبيا سبا كذا
أرادني و هــــن لا ذي افتح (شذا)
78 عــــــهدي و ربي مع آياتي و في
آتاني الكــــــتاب عنه افتح تفي
79 و في الــندا افتح (جاد) بالخلف ( فني)
نعمـــــــتي التي فزد مع جاءني
80 البـــــينات فاسكنن (حبر) (مهر)
بلغـــــــني أروني الذين ( مر)
81 (طب) حسـبي المكي و الأخري ( جلا)
مع شــــــــركائي الذين أولا
82 و عنه باقــــــي الباب بالخلاف
كمــــــسني بالحجر و الأعراف
83 و عند همز الوصــل (فز ) أخي سكن
قـومي و بعدي ( مز) و غير ذي حسن
84 و مع ســـــوي همز له افتح و لي
دين و للمــــــكي بإسكان جلي
85 و في صراطي اشـرح لي افتحن ( حجا)
و هكذا قومــــــي  ليلا عنه جا
86 و في أخي معا و نفــــــسي أولا
لدي العقود فتحـــــهن ( حصلا)
*******************
باب مذاهبهم في ياءات الزوائد
87 أثبت يدع الــــداع (مز) دعاء مع
أكرمن أهانن وصـــــــلا (جمع)
88 و اثبــــتهما (حلا) و حذفهن ( فن)
آتان ( حز) بالــــــواد عنه اثبتن
89 و اتبعون زخـــــرف حاليه (فج)
و في رءوس الآي حال الوصــل (حج)
90 ثم اليزيــــــدي كأبي عمرو سوا
فيما علـــــيه ذلك الباب احتوي
91 بشر عـــــــبادي يتقي يرتع لهم
فاحـــــذف و قد تمت هنا أصولهم
*******************
باب فرش الحروف سورة البقرة
92 لاريب بالتــــنوين حيث جا (حلا)
أنذرتهم مـــــــعا بإخبار (ملا)
93 غشاوة فاضـــــمم او افتح معجما
وفيه ضـــــــم مع إهمال (حما)
94 ويخــــدعون (من) (حميد) و (حتم)
قل يكــــــذبون قيل الست أشم
95 (حز) (شم) و سـئ سيئت الخلف (جنا)
يمد ضـــــم اكسر (فتي ) و أسكنا
96 بحــــــيث ظلمات من الصواقع
قل و الـــــصواقع بذرو(حز) تعي
97 خا يخطف افتح (طاب) و اكسـرها (حما)
مع يا و شد الطاء و اكسر عنهـــما
98 و يستـــحي (ماضي) و كيف يرجع
فسم ( من) (طب) للأخري راجـــع
99 و في قد أفلح (منا) (طـــب) (حصلا)
مع ترجع الأمور حيث أنــــــزلا
100 كذاك في أول قـــــــص و بذا
في يرجع الأمر الجميــــــع أخذا
101 علم ضـــم اكسر و بعد ارفع (حفا)
قبل اســـجدوا اضمم الملائكة (شفا)
102 وصل بلا ها من كــــهذي الشجرة
إلا التي من بعد يحيي (مبـــــصرة)
103 وهذه الحق فاثبــــــــــتنها
لا خوف  للمـــــكي دع تنوينها
104 وحسن كالحضـــــرمي و إسرئل
له و بين بين (طب) حيـــــث يحل
105 يقبل ذكر (حز) ويذبحـــــون مع
يذبح مكي وعـــــدنا اقصر (جمع)
106 لا (حز) ورب في الندا يا قـــوم ضم
من قبل همز الوصــل (فز) و (جد) يعم
107 بارئكم له اختلـــــس كذا اسكنن
في باب يامركم و نــطعمكم  و (فن)
108 فأخف و الغير لـــــــكل أكملا
و الصعقة اقرأ (مــز) و في ذرو (جلا)
109 و( حز) خطـــــيئاتكم رجزا بضم
نصبا و جرا عنـــــد تنوين (معم)
110 و حيث يفـــسقون بالكسر ( اتصف)
عشرة عينا (طب) و في الأخري اختلف
111 و لاتنوين مــــصر ( حائز) ( العلا)
و اذكروا ( طوي) افتح اشدد مسـجلا
112 هزؤا و كفؤا ضــــم مبدلا (شذا)
و متشــــــــابه علينا ( حبذا)
113 يشابه المــــــــطوعي اشدد لما
مع خلف الآخــــرين يهبط اضمما
114 و كلم اقرأ عنــــده خاطب (مضا)
لا تعلمــــــون و معا بعد (فضا)
115 خف الأماني و أماني للحــــــسن
و الرفع و الجر اســكنن و الها اكسرن
116 و يعبـــــدون الغيب (حامد) و لا
تنوين في حســـنا و قل أسري (حلا)
117 تقتلون اشــــــدد مع الثالث ثم
تظهرون القــــصر و التشديد (حم)
118 وقل تفــــادوهم (منا) (طب) امددا
و خففن للمـــــــك كيف أيدا
119 و الرسل ســكن كيف جا (حز) وافقه
في غير ما به ضــــــمير (طابقه)
120 و رسلنا مع هم و كـــــم بصريهم
عقبا و حقبا (حز) و خـــبرا عنه ضم
121 خشــب و عرفا عذرا او نذرا (حكوا)
عربا بضمــــهم هنا غلف ( مضوا)
